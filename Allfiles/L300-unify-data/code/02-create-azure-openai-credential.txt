CREATE THE AZURE OPENAI DATABASE CREDENTIAL
Language: T-SQL

-- Replace <azure-openai-resource> with the Azure OpenAI resource name.
-- The Fabric SQL managed identity must have the Cognitive Services OpenAI User role.

IF EXISTS (
    SELECT 1 FROM sys.database_scoped_credentials
    WHERE name = 'https://<azure-openai-resource>.openai.azure.com'
)
    DROP DATABASE SCOPED CREDENTIAL [https://<azure-openai-resource>.openai.azure.com];
GO

CREATE DATABASE SCOPED CREDENTIAL [https://<azure-openai-resource>.openai.azure.com]
WITH IDENTITY = 'Managed Identity',
     SECRET = '{"resourceid":"https://cognitiveservices.azure.com"}';
GO
